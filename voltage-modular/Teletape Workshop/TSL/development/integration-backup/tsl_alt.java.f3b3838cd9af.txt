    private double pulseTime = 0.0;

    // Constants derived from EMS Sequencer 256 physical constraints
    private static final double MAX_SPEED = 1000.0; // 1000 V/s (fastest rate)
    private static final double MIN_SPEED = 0.05;   // 0.05 V/s (slowest rate)
    private static final double LOG2_MIN_SLEW = Math.log(MIN_SPEED) / Math.log(2.0);
    private static final double LOG2_MAX_SLEW = Math.log(MAX_SPEED) / Math.log(2.0);
    private static final double LOG2_SLEW_SPAN = LOG2_MAX_SLEW - LOG2_MIN_SLEW;
    private static final double SAMPLE_PERIOD = 1.0 / 48000.0;

    // Safe mathematical boundaries handler
    private static double clamp(double value, double min, double max) {
        return Math.min(max, Math.max(min, value));
    }

    // Input signal validation engine
    private double readInput(VoltageAudioJack jack) {
        if (jack.IsConnected()) {
            double v = jack.GetValue();
            return Double.isFinite(v) ? v : 0.0;
        }
        return 0.0;
    }

    // 1V/Octave Log-Exponential Translation Engine
    private static double rateFor(double knobValue, double controlVoltage) {
        double position = clamp(knobValue, 0.0, 1.0);
        double baseLog2Rate = LOG2_MIN_SLEW + (position * LOG2_SLEW_SPAN);
        double cv = Double.isFinite(controlVoltage) ? controlVoltage : 0.0;
        double cvRateShift = cv * 1.0; // Precise 1 Volt per Octave scaling shift
        double modulatedLog2Rate = clamp(baseLog2Rate + cvRateShift, LOG2_MIN_SLEW, LOG2_MAX_SLEW);
        return Math.pow(2.0, modulatedLog2Rate);
    }

    @Override
    public void Initialize()
    {
        // Start the continuous graphical clock frame loop
        StartGUITimer();
        lampPower.SetValue( 1.0 );

        // --- Performance Testing Block ---
        final int iterations = 10000000; 
        double dummyAccumulator = 0.0;
        long startTime = System.nanoTime();
for( int i = 0; i < iterations; i++ )
{
double mockCV = ( i % 2 == 0 ) ? 1.5 : -0.5;
double mockKnob = ( i % 100 ) / 100.0;
double rateSample = rateFor( mockKnob, mockCV );
dummyAccumulator += rateSample;
}
long endTime = System.nanoTime();
long totalDurationNs = endTime - startTime;
double totalDurationMs = totalDurationNs / 1000000.0;
double averageTimePerSampleNs = (double) totalDurationNs / iterations;
System.out.println( String.format( "Result: Processed %d cycles in %.2f ms", iterations, totalDurationMs ) );
System.out.println( String.format( "Average Cost: %.2f nanoseconds per sample", averageTimePerSampleNs ) );
if( dummyAccumulator == 42.42 ) {
System.out.println( "Check: " + dummyAccumulator );
}
System.out.println( "=== TSL PERFORMANCE BENCHMARK COMPLETE ===" );
}

@Override
public void ProcessSample()
{
//[user-ProcessSample] Integrated EMS Slew / Track & Hold Engine
double dt = SAMPLE_PERIOD;
// If a preset was loaded, break out of standard integration math and synchronize immediately
if ( resumePending )
{
currentVolt1 = readInput( inputV1 );
currentVolt2 = readInput( inputV2 );
currentVolt3 = readInput( inputV3 );
resumePending = false;
}
// --- CHANNEL 1 (Standard EMS Slew) ---
double input1 = readInput( inputV1 );
double cv1 = inputVC1.IsConnected() ? inputVC1.GetValue() : 0.0;
double maxRate1 = rateFor( knobV1.GetValue(), cv1 );
double maxChange1 = maxRate1 * dt;
double diff1 = input1 - currentVolt1;
if (diff1 > maxChange1)       currentVolt1 += maxChange1;
else if (diff1 < -maxChange1) currentVolt1 -= maxChange1;
else                          currentVolt1 = input1;
outputV1.SetValue( currentVolt1 );
// --- CHANNEL 2 (Standard EMS Slew) ---
double input2 = readInput( inputV2 );
double cv2 = inputVC2.IsConnected() ? inputVC2.GetValue() : 0.0;
double maxRate2 = rateFor( knobV2.GetValue(), cv2 );
double maxChange2 = maxRate2 * dt;
double diff2 = input2 - currentVolt2;
if (diff2 > maxChange2)       currentVolt2 += maxChange2;
else if (diff2 < -maxChange2) currentVolt2 -= maxChange2;
else                          currentVolt2 = input2;
outputV2.SetValue( currentVolt2 );
// --- CHANNEL 3 (Dual Identity: Slew or Track & Hold) ---
double input3 = readInput( inputV3 );
// Dynamic Gate/Clock Connector Check
boolean trackMode3 = !inputVC3.IsConnected() || (inputVC3.GetValue() > 1.0);
if (trackMode3)
{
// Set CV parameter to zero during gated tracks to isolate knob motion limits
double cvParam3 = 0.0;
double maxRate3 = rateFor( knobV3.GetValue(), cvParam3 );
double maxChange3 = maxRate3 * dt;
double diff3 = input3 - currentVolt3;
if (diff3 > maxChange3)       currentVolt3 += maxChange3;
else if (diff3 < -maxChange3) currentVolt3 -= maxChange3;
else                          currentVolt3 = input3;
}
else
{
// Integrator path is completely open. Hold last value.
}
outputV3.SetValue( currentVolt3 );
//[/user-ProcessSample]
}
@Override
public void ProcessBypassedSample()
{
//[user-ProcessBypassedSample] Continuous Tracking Bypass
currentVolt1 = readInput( inputV1 );
currentVolt2 = readInput( inputV2 );
currentVolt3 = readInput( inputV3 );
outputV1.SetValue( currentVolt1 );
outputV2.SetValue( currentVolt2 );
outputV3.SetValue( currentVolt3 );
//[/user-ProcessBypassedSample]
}
@Override
public String GetTooltipText( VoltageComponent component )
{
//[user-GetTooltipText] Real-time hardware rate calculation display
if ( component == knobV1 || component == knobV2 || component == knobV3 )
{
VoltageKnob knob = (VoltageKnob) component;
double currentRate = rateFor( knob.GetValue(), 0.0 );
if ( currentRate >= 10.0 ) {
return String.format( "Slew Rate: %.1f V/s", currentRate );
} else {
return String.format( "Slew Rate: %.3f V/s", currentRate );
}
}
return super.GetTooltipText( component );
//[/user-GetTooltipText]
}
@Override
public void EditComponentValue( VoltageComponent component, double newValue, String newText )
{
//[user-EditComponentValue] Parse manual user text input overrides
if ( component == knobV1 || component == knobV2 || component == knobV3 )
{
try
{
String cleanText = newText.replaceAll( "(?i)[a-z:/\s]", "" );
double typedRate = Double.parseDouble( cleanText );
typedRate = clamp( typedRate, MIN_SPEED, MAX_SPEED );
double log2Typed = Math.log( typedRate ) / Math.log( 2.0 );
double targetKnobPos = ( log2Typed - LOG2_MIN_SLEW ) / LOG2_SLEW_SPAN;
targetKnobPos = clamp( targetKnobPos, 0.0, 1.0 );
VoltageKnob knob = (VoltageKnob) component;
knob.SetValue( targetKnobPos );
Notify( knob, ModuleNotifications.Knob_Changed, targetKnobPos, 0, 0, 0, null );
return;
}
catch ( Exception e ) {
// Fallback on error
}
}
super.EditComponentValue( component, newValue, newText );
//[/user-EditComponentValue]
}

@Override
public byte[] GetStateInformation()
{
//[user-GetStateInformation] Save current settings into preset files
try
{
java.io.ByteArrayOutputStream bos = new java.io.ByteArrayOutputStream();
java.io.ObjectOutputStream oos = new java.io.ObjectOutputStream(bos);
oos.writeDouble(currentVolt1);
oos.writeDouble(currentVolt2);
oos.writeDouble(currentVolt3);
oos.flush();
return bos.toByteArray();
}
catch (Exception e) {
return null;
}
//[/user-GetStateInformation]
}
@Override
public void SetStateInformation(byte[] stateInfo)
{
//[user-SetStateInformation] Recall settings seamlessly when loading presets
if (stateInfo == null || stateInfo.length == 0) return;
try
{
java.io.ByteArrayInputStream bis = new java.io.ByteArrayInputStream(stateInfo);
java.io.ObjectInputStream ois = new java.io.ObjectInputStream(bis);
currentVolt1 = ois.readDouble();
currentVolt2 = ois.readDouble();
currentVolt3 = ois.readDouble();
resumePending = true;
}
catch (Exception e) {
// Keep current states on exception
}
//[/user-SetStateInformation]
}
@Override
public byte[] GetStateInformationForVariations()
{
return GetStateInformation();
}

public void SetStateInformationForVariations(byte[] stateInfo)
{
